def init_db():
    conn = sqlite3.connect('queue.db')
    c = conn.cursor()
    
    # Таблица с пользователями
    c.execute('''CREATE TABLE IF NOT EXISTS users
                 (user_id INTEGER PRIMARY KEY,
                  username TEXT,
                  full_name TEXT,
                  registered_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)''')
    
    # Таблица с бронированиями
    c.execute('''CREATE TABLE IF NOT EXISTS bookings
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  user_id INTEGER,
                  break_type TEXT,
                  start_time TEXT,
                  end_time TEXT,
                  status TEXT DEFAULT 'active',
                  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                  FOREIGN KEY (user_id) REFERENCES users (user_id))''')
    
    conn.commit()
    conn.close()
    print("✅ База данных инициализирована")

def register_user(user_id, username, full_name):
    """Регистрация пользователя в БД"""
    conn = sqlite3.connect('queue.db')
    c = conn.cursor()
    try:
        c.execute('''INSERT OR IGNORE INTO users (user_id, username, full_name)
                     VALUES (?, ?, ?)''', (user_id, username, full_name))
        conn.commit()
    finally:
        conn.close()

def get_current_bookings_count(break_type, start_time):
    """Сколько человек уже записались на этот слот"""
    conn = sqlite3.connect('queue.db')
    c = conn.cursor()
    c.execute('''SELECT COUNT(*) FROM bookings 
                 WHERE break_type = ? AND start_time = ? AND status = 'active' ''',
              (break_type, start_time))
    count = c.fetchone()[0]
    conn.close()
    return count

def get_user_active_bookings(user_id):
    """Активные брони пользователя"""
    conn = sqlite3.connect('queue.db')
    c = conn.cursor()
    c.execute('''SELECT break_type, start_time, end_time FROM bookings 
                 WHERE user_id = ? AND status = 'active' 
                 ORDER BY start_time''', (user_id,))
    bookings = c.fetchall()
    conn.close()
    return bookings

def create_booking(user_id, break_type, start_time, duration_minutes):
    """Создание бронирования"""
    # Преобразуем время в формат ЧЧ:ММ
    start_dt = datetime.strptime(start_time, "%H:%M")
    end_dt = start_dt + timedelta(minutes=duration_minutes)
    
    # Если переходит через полночь
    if end_dt.day > start_dt.day:
        end_time = end_dt.strftime("%H:%M") + " (след. день)"
    else:
        end_time = end_dt.strftime("%H:%M")
    
    conn = sqlite3.connect('queue.db')
    c = conn.cursor()
    
    # Проверяем лимит
    current_count = get_current_bookings_count(break_type.value[0], start_time)
    max_users = break_type.value[2]
    
    if current_count >= max_users:
        conn.close()
        return False, f"❌ На {break_type.value[0]} {start_time} уже записалось {current_count}/{max_users} человек"
    
    # Проверяем, нет ли у пользователя уже брони на этот тип перерыва
    c.execute('''SELECT start_time FROM bookings 
                 WHERE user_id = ? AND break_type = ? AND status = 'active' ''',
              (user_id, break_type.value[0]))
    existing = c.fetchone()
    
    if existing:
        conn.close()
        return False, f"❌ У вас уже есть активный {break_type.value[0]} на {existing[0]}"
    
    # Создаем бронь
    try:
        c.execute('''INSERT INTO bookings (user_id, break_type, start_time, end_time)
                     VALUES (?, ?, ?, ?)''',
                  (user_id, break_type.value[0], start_time, end_time))
        conn.commit()
        return True, f"✅ Вы записались на {break_type.value[0]}\n⏰ Время: {start_time}-{end_time}"
    except Exception as e:
        return False, f"❌ Ошибка: {str(e)}"
    finally:
        conn.close()

def cancel_booking_by_type(user_id, break_type):
    """Отменить конкретное бронирование пользователя"""
    conn = sqlite3.connect('queue.db')
    c = conn.cursor()
    c.execute('''UPDATE bookings SET status = 'cancelled' 
                 WHERE user_id = ? AND break_type = ? AND status = 'active' ''',
              (user_id, break_type.value[0]))
    deleted = c.rowcount
    conn.commit()
    conn.close()
    return deleted

def get_all_bookings():
    """Все активные бронирования"""
    conn = sqlite3.connect('queue.db')
    c = conn.cursor()
    c.execute('''SELECT b.break_type, b.start_time, b.end_time, u.username, u.full_name
                 FROM bookings b
                 JOIN users u ON b.user_id = u.user_id
                 WHERE b.status = 'active'
                 ORDER BY 
                   CASE 
                     WHEN length(b.start_time) = 5 THEN b.start_time
                     ELSE '23:59'
                   END,
                   b.break_type''')
    bookings = c.fetchall()
    conn.close()
    return bookings

def generate_time_slots(interval_minutes=15):
    """Генерация временных слотов на 24 часа"""
    slots = []
    for hour in range(24):  # 0-23 часов
        for minute in range(0, 60, interval_minutes):
            time_str = f"{hour:02d}:{minute:02d}"
            slots.append(time_str)
    return slots

            message += f"{i}. ⏰ {start_time}-{end_time}\n"
            message += f"   👤 {user_display} ({current_count}/5)\n"

        message += "🚬 *Перекуры (10 мин):*\n"
        smoke_bookings.sort(key=lambda x: x[0])

async def show_time_selection(update: Update, context: ContextTypes.DEFAULT_TYPE, break_type: BreakType):
    """Показать выбор времени (всплывающее меню)"""
    now = datetime.now()
    current_time = now.strftime("%H:%M")
    
    # Генерируем слоты на ближайшие 4 часа
    all_slots = []
    start_hour = now.hour
    end_hour = (start_hour + 4) % 24
    
    for hour_offset in range(4):
        hour = (start_hour + hour_offset) % 24
        for minute in [0, 15, 30, 45]:
            time_str = f"{hour:02d}:{minute:02d}"
            all_slots.append(time_str)
    
    # Создаем инлайн-кнопки (всплывающее меню)
    keyboard = []
    row = []
    
    for i, start_time in enumerate(all_slots, 1):
        # Проверяем сколько уже записалось
        current_count = get_current_bookings_count(break_type.value[0], start_time)
        max_users = break_type.value[2]
        
        # Рассчитываем время окончания
        start_dt = datetime.strptime(start_time, "%H:%M")
        end_dt = start_dt + timedelta(minutes=break_type.value[1])
        
        if end_dt.day > start_dt.day:
            display_time = f"{start_time}⁺¹"
        else:
            display_time = start_time
        
        if current_count >= max_users:
            # Слот заполнен
            btn_text = f"🔴 {display_time}"
            callback_data = f"full_{break_type.value[0].replace(' ', '_')}_{start_time}"
        else:
            # Есть свободные места
            btn_text = f"🟢 {display_time}"
            callback_data = f"book_{break_type.value[0].replace(' ', '_')}_{start_time}"
        
        row.append(InlineKeyboardButton(btn_text, callback_data=callback_data))
        
        # 4 кнопки в ряд
        if i % 4 == 0:
            keyboard.append(row)
            row = []
    
    if row:
        keyboard.append(row)
    
    # Кнопки навигации
    keyboard.append([
        InlineKeyboardButton("🔄 Больше времени", callback_data=f"more_time_{break_type.value[0].replace(' ', '_')}"),
        InlineKeyboardButton("🔙 Назад", callback_data="back_to_main")
    ])
    
    current_count = get_current_bookings_count(break_type.value[0], current_time[:5])
    
    await update.message.reply_text(
        f"*{break_type.value[0]} - Выберите время:*\n\n"
        f"🟢 - есть свободные места\n"
        f"🔴 - все места заняты\n"
        f"👥 - до {max_users} человек одновременно\n\n"
        f"⏰ *Текущее время:* {current_time}\n"
        f"📊 *Сейчас записано:* {current_count}/{max_users}\n\n"
        f"📅 *Ближайшие 4 часа:*",
        reply_markup=InlineKeyboardMarkup(keyboard),
        parse_mode='Markdown'
    )

    else:
        break_type = BreakType.SMOKE
        break_type_display = "🚬 Перекур"

    }

        elif hour < 12:

        else:
            time_blocks["🌙 Ночь (18:00-24:00)"].append(time_str)

                f"✅ Перекур отменен ({deleted} запись)",
                reply_markup=get_main_keyboard()

        else:
            await update.message.reply_text(
                "❌ У вас нет активных записей на перекур",
                reply_markup=get_main_keyboard()
            )
    
    elif text == "Отменить всё":
        conn = sqlite3.connect('queue.db')
        c = conn.cursor()
        c.execute('''UPDATE bookings SET status = 'cancelled' 
                     WHERE user_id = ? AND status = 'active' ''', (user.id,))
        deleted = c.rowcount
        conn.commit()
        conn.close()
        
        if deleted > 0:
            await update.message.reply_text(
                f"✅ Отменены все записи ({deleted})",
                reply_markup=get_main_keyboard()
            )
        else:

async def show_stats(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Показать статистику"""
    conn = sqlite3.connect('queue.db')
    c = conn.cursor()
    
    # Общая статистика
    c.execute('''SELECT break_type, COUNT(*) as count FROM bookings 
                 WHERE status = 'active' GROUP BY break_type''')
    active_stats = dict(c.fetchall())
    
    c.execute('''SELECT COUNT(DISTINCT user_id) FROM users''')
    total_users = c.fetchone()[0]

    message = "📊 *Статистика использования бота:*\n\n"

    message += f"• 👥 Всего активных записей: {total_active}\n\n"
    
    message += f"👥 Всего пользователей: {total_users}\n"
    message += f"⏰ Текущее время: {datetime.now().strftime('%H:%M')}\n\n"
    
    message += "*📌 Быстрые действия:*\n"
    message += "Используйте кнопки ниже 👇"
    
    await update.message.reply_text(
        message,
        reply_markup=get_main_keyboard(),
        parse_mode='Markdown'
    )

async def button_callback(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Обработка инлайн-кнопок (всплывающих меню)"""
    query = update.callback_query
    await query.answer()
    
    data = query.data
    user = query.from_user
    
    # Бронирование времени
    if data.startswith("book_"):
        # Формат: book_Тип_Время
        parts = data.split("_")
        if len(parts) >= 4:
            break_type_str = parts[1] + " " + parts[2]  # Эмодзи + тип
            start_time = parts[3]
            
            # Определяем тип перерыва
            if "Обед" in break_type_str:
                break_type = BreakType.LUNCH
            else:
                break_type = BreakType.SMOKE
            
            # Создаем бронь
            success, message = create_booking(user.id, break_type, start_time, break_type.value[1])
            
            if success:
                # Получаем обновленную информацию о слоте
                current_count = get_current_bookings_count(break_type.value[0], start_time)
                max_users = break_type.value[2]
                
                # Рассчитываем время окончания
                start_dt = datetime.strptime(start_time, "%H:%M")
                end_dt = start_dt + timedelta(minutes=break_type.value[1])
                
                if end_dt.day > start_dt.day:
                    end_display = end_dt.strftime("%H:%M") + " (следующий день)"
                else:
                    end_display = end_dt.strftime("%H:%M")
                
                await query.edit_message_text(
                    f"✅ *Запись подтверждена!*\n\n"
                    f"📋 *Тип:* {break_type.value[0]}\n"
                    f"👤 *Имя:* {user.first_name}\n"
                    f"⏰ *Время:* {start_time} - {end_display}\n"
                    f"⏱ *Длительность:* {break_type.value[1]} минут\n"
                    f"👥 *Записано на это время:* {current_count}/{max_users} человек\n\n"
                    f"📌 *Запись автоматически снимется через {break_type.value[1]} минут*",
                    parse_mode='Markdown'
                )
                
                # Планируем автоотмену
                context.job_queue.run_once(
                    auto_cancel_job,
                    break_type.value[1] * 60,
                    data={
                        "user_id": user.id, 
                        "break_type": break_type.value[0], 
                        "start_time": start_time,
                        "user_name": user.first_name
                    }
                )
            else:
                await query.edit_message_text(
                    message + "\n\nПопробуйте другое время:",
                    reply_markup=InlineKeyboardMarkup([[InlineKeyboardButton("🔄 Выбрать другое время", callback_data=f"select_{parts[1]}_{parts[2]}")]])
                )
    
    # Информация о заполненных слотах
    elif data.startswith("full_"):
        parts = data.split("_")
        if len(parts) >= 4:
            break_type_str = parts[1] + " " + parts[2]
            start_time = parts[3]
            current_count = get_current_bookings_count(break_type_str, start_time)
            
            if "Обед" in break_type_str:
                max_users = BreakType.LUNCH.value[2]
            else:
                max_users = BreakType.SMOKE.value[2]
            
            await query.answer(f"❌ Этот слот заполнен ({current_count}/{max_users})", show_alert=True)
    
    # Выбор типа перерыва
    elif data.startswith("select_"):
        parts = data.split("_")
        if len(parts) >= 3:
            break_type_str = parts[1] + " " + parts[2]
            if "Обед" in break_type_str:
                await show_time_selection(query, context, BreakType.LUNCH)
            else:
                await show_time_selection(query, context, BreakType.SMOKE)
    
    # Больше времени
    elif data.startswith("more_time_"):
        parts = data.split("_")
        if len(parts) >= 3:
            break_type_str = parts[1] + " " + parts[2]
            await show_more_time_options(update, context, break_type_str)
    
    # Назад в главное меню
    elif data in ["back_to_main", "back_to_main_menu"]:
        await query.edit_message_text(
            "Главное меню 👇\nИспользуйте кнопки ниже:",
            reply_markup=get_main_keyboard()
        )

async def auto_cancel_job(context: ContextTypes.DEFAULT_TYPE):
    """Автоматическая отмена брони через время перерыва"""
    data = context.job.data
    user_id = data["user_id"]
    break_type = data["break_type"]
    start_time = data["start_time"]
    user_name = data["user_name"]
    
    conn = sqlite3.connect('queue.db')
    c = conn.cursor()
    c.execute('''UPDATE bookings SET status = 'completed' 
                 WHERE user_id = ? AND break_type = ? AND start_time = ? AND status = 'active' ''',
              (user_id, break_type, start_time))
    rows_updated = c.rowcount
    conn.commit()
    conn.close()
    
    if rows_updated > 0:
        print(f"🔄 {break_type} {start_time} завершен для {user_name} (ID: {user_id})")
